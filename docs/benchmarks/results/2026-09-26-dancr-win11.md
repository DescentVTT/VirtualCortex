## The script

```bash
#!/usr/bin/env bash
# usage: pairs.sh <base active binary> <change active binary> <output file>
base_bin=$1
change_bin=$2
out=$3
raw="$out.d"
tests=(
  the_active_set_under_the_reference_drive_at_1024_units_exhaustive
  the_active_set_under_a_drive_sixteen_times_sparser_at_1024_units_exhaustive
  the_active_set_under_a_drive_256_times_sparser_at_1024_units_exhaustive
  the_active_set_under_the_reference_drive_at_4096_units_exhaustive
)
run_build() {
  local label=$1 bin=$2 pair=$3
  for k in 0 1 2 3; do
    local log="$raw/$label-$pair-$k.log"
    "$bin" --ignored --exact "${tests[$k]}" --nocapture --test-threads=1 >"$log" 2>&1
    if ! grep -q "test result: ok. 1 passed" "$log"; then
      echo "$label pair=$pair run=$k FAILED" >>"$out"
    fi
    grep -oE "DUMP (network|control) [0-9] \[.* ns_per_tick [0-9]+ worker_ns_per_turn [0-9]+" "$log" |
      sed -E "s/^DUMP (network|control) ([0-9]) .* ns_per_tick ([0-9]+) worker_ns_per_turn ([0-9]+)$/$label $pair \1 \2 \3 \4/" >>"$out"
    grep -oE "DUMP (network|control) [0-9] turns per worker .*" "$log" |
      sed -E "s/^DUMP (network|control) ([0-9]) turns per worker (.*)/$label $pair \1 \2 shares \3/" >>"$out"
  done
}
: >"$out"
mkdir -p "$raw"
echo "start $(date -u +%FT%TZ)" >>"$out"
for pair in 1 2 3 4 5; do
  if ((pair % 2 == 1)); then
    run_build base "$base_bin" "$pair"
    run_build change "$change_bin" "$pair"
  else
    run_build change "$change_bin" "$pair"
    run_build base "$base_bin" "$pair"
  fi
done
echo "end $(date -u +%FT%TZ)" >>"$out"
```

The medians are the third of each run's five readings sorted; the ratio is the change's median over the base's.
