```js
// usage: node analyse.mjs disturbance <out dir>
//        node analyse.mjs readings <out dir> <part>...
// `disturbance` applies ADR-0102's rule to load.tsv and the parts' windows in events.txt and
// reads nothing else; `readings` takes each run's median, the third of its five readings
// sorted, and the ratio of the change's median to the base's.
import fs from "node:fs";
import path from "node:path";

const [mode, out, ...parts] = process.argv.slice(2);
const events = fs.readFileSync(path.join(out, "events.txt"), "utf8").split(/\r?\n/);

if (mode === "disturbance") {
  const window = {};
  for (const line of events) {
    const m = line.match(/^part (\S+) (start|end) (\S+)$/);
    if (m) (window[m[1]] ??= {})[m[2]] = Date.parse(m[3]);
  }
  const load = fs
    .readFileSync(path.join(out, "load.tsv"), "utf8")
    .split(/\r?\n/)
    .slice(1)
    .filter(Boolean)
    .map((l) => {
      const [t, total, round, others, top] = l.split("\t");
      return { t: Date.parse(t), total: +total, round: +round, others: +others, top };
    });
  // A sample covers the five seconds before its stamp; it is in a part if it overlaps it.
  const LIMIT = 4; // logical processors held by processes outside the round's binaries
  const RUN = 3; // consecutive samples
  function judge(part) {
    const w = window[part];
    if (!w || !w.end) return { part, complete: false };
    const s = load.filter((x) => x.t > w.start && x.t - 5000 < w.end);
    let run = 0, worst = 0, at = null;
    for (const x of s) {
      run = x.others > LIMIT ? run + 1 : 0;
      if (run > worst) { worst = run; at = x; }
    }
    const mean = (a) => a.reduce((p, q) => p + q, 0) / (a.length || 1);
    const totals = s.map((x) => x.total).sort((a, b) => a - b);
    const top = s.reduce((best, x) => (x.others > (best?.others ?? -1) ? x : best), null);
    return {
      part, complete: true, start: new Date(w.start).toISOString(), end: new Date(w.end).toISOString(),
      samples: s.length, disturbed: worst >= RUN, longestRunAboveLimit: worst, where: at ? `${new Date(at.t).toISOString()} ${at.top}` : null,
      total: { mean: +mean(totals).toFixed(1), max: totals.at(-1), atOrAbove40: totals.filter((v) => v >= 40).length },
      others: { mean: +mean(s.map((x) => x.others)).toFixed(2), max: Math.max(...s.map((x) => x.others)), heaviest: top ? top.top : null },
      round: { mean: +mean(s.map((x) => x.round)).toFixed(2) },
    };
  }
  for (const part of Object.keys(window)) console.log(JSON.stringify(judge(part)));
} else if (mode === "readings") {
  for (const part of parts) {
    const rows = fs
      .readFileSync(path.join(out, `${part}.tsv`), "utf8")
      .split(/\r?\n/)
      .filter(Boolean)
      .map((l) => l.split("\t"));
    const failed = rows.filter((r) => r[4] === "FAILED");
    const runs = {};
    for (const r of rows) {
      if (r[4] === "FAILED" || r[6] === "shares") continue;
      const [, label, pair, i, kind, k, ns, perTurn] = r;
      const key = `${i} ${kind}`;
      ((runs[key] ??= { base: [], change: [] })[label]).push({ pair: +pair, ns: +ns, perTurn: +perTurn, k });
    }
    const median = (a) => [...a].sort((x, y) => x - y)[Math.floor(a.length / 2)];
    console.log(`part ${part}: ${failed.length} failed runs`);
    for (const key of Object.keys(runs).sort()) {
      const { base, change } = runs[key];
      const order = (a) => [...a].sort((x, y) => x.pair - y.pair);
      const b = order(base), c = order(change);
      const mb = median(b.map((x) => x.ns)), mc = median(c.map((x) => x.ns));
      console.log(
        [
          part, key, `k=${b[0]?.k}`,
          `base ${b.map((x) => x.ns).join(", ")} median ${mb}`,
          `change ${c.map((x) => x.ns).join(", ")} median ${mc}`,
          `ratio ${(mc / mb).toFixed(3)}`,
          `per turn base ${median(b.map((x) => x.perTurn))} change ${median(c.map((x) => x.perTurn))}`,
          `n ${b.length}/${c.length}`,
        ].join(" | "),
      );
    }
  }
}
```

```rust
mod aos;
use cortex_core::{DendriticSuperNeuron, MembraneLanes, LANES};
use std::hint::black_box;
use std::time::Instant;

fn load(path: &str) -> Vec<DendriticSuperNeuron> {
    let text = std::fs::read_to_string(path).unwrap();
    text.lines().enumerate().map(|(i, l)| {
        let f: Vec<i64> = l.split(' ').map(|x| x.parse().unwrap()).collect();
        let mut u = DendriticSuperNeuron::new(i as u64);
        u.v_soma = f[0] as i32; u.v_basal = f[1] as i32; u.v_apical = f[2] as i32; u.v_thresh = f[3] as i32;
        u.bac_plateau_ticks = f[4] as u16; u.refractory_ticks = f[5] as u16; u.flags = f[6] as u8; u.last_soma_spike_tick = f[7] as u32;
        u
    }).collect()
}
fn reset(dst: &mut [DendriticSuperNeuron], src: &[DendriticSuperNeuron]) {
    for (d, s) in dst.iter_mut().zip(src) {
        d.v_soma = s.v_soma; d.v_basal = s.v_basal; d.v_apical = s.v_apical; d.v_thresh = s.v_thresh;
        d.bac_plateau_ticks = s.bac_plateau_ticks; d.refractory_ticks = s.refractory_ticks; d.flags = s.flags;
    }
}
const TICKS: u32 = 64;
const ROUNDS: u32 = 2000;
fn main() {
    let snap = load(&std::env::args().nth(1).unwrap());
    let mut units = load(&std::env::args().nth(1).unwrap());
    let n = units.len() as f64;
    for pass in 0..3 {
        // scalar
        let mut total = 0u128; let mut fired = 0u32;
        for r in 0..ROUNDS { reset(&mut units, &snap); let t0 = Instant::now();
            for t in 0..TICKS { let now = r * TICKS + t; for u in units.iter_mut() { fired += u.integrate(black_box(0), black_box(0), now) as u32; } }
            total += t0.elapsed().as_nanos(); }
        let scalar = total as f64 / (ROUNDS as f64 * TICKS as f64 * n);
        // lanes with load/store
        let mut lanes = MembraneLanes::new(); let mut total = 0u128; let mut fired2 = 0u32;
        for r in 0..ROUNDS { reset(&mut units, &snap); let t0 = Instant::now();
            for t in 0..TICKS { let now = r * TICKS + t;
                for chunk in units.chunks_mut(LANES) {
                    for (lane, u) in chunk.iter().enumerate() { lanes.load(lane, u, black_box(0), black_box(0)); }
                    lanes.integrate();
                    for (lane, u) in chunk.iter_mut().enumerate() { fired2 += lanes.store(lane, u, now) as u32; }
                } }
            total += t0.elapsed().as_nanos(); }
        let full = total as f64 / (ROUNDS as f64 * TICKS as f64 * n);
        let mut aosl = aos::Aos::new(); let mut total = 0u128; let mut fired3 = 0u32;
        for r in 0..ROUNDS { reset(&mut units, &snap); let t0 = Instant::now();
            for t in 0..TICKS { let now = r * TICKS + t;
                for chunk in units.chunks_mut(aos::N) { let mut aosl = aos::Aos::new();
                    for (lane, u) in chunk.iter().enumerate() { aosl.load(lane, u, black_box(0), black_box(0)); }
                    aosl.integrate();
                    for (lane, u) in chunk.iter_mut().enumerate() { fired3 += aosl.store(lane, u, now) as u32; }
                } }
            total += t0.elapsed().as_nanos(); }
        let aosfull = total as f64 / (ROUNDS as f64 * TICKS as f64 * n);
        println!("  aos lanes+load/store {aosfull:.2} (fired {fired3})");
        // lanes integrate only
        let mut many = vec![MembraneLanes::new(); units.len() / LANES];
        for (c, chunk) in units.chunks(LANES).enumerate() { for (lane, u) in chunk.iter().enumerate() { many[c].load(lane, u, 0, 0); } }
        let t0 = Instant::now();
        for _ in 0..(ROUNDS * TICKS / 8) { for l in many.iter_mut() { l.integrate(); } black_box(&many); }
        let only = t0.elapsed().as_nanos() as f64 / ((ROUNDS * TICKS / 8) as f64 * n);
        println!("pass {pass}: scalar {scalar:.2} ns/unit, lanes+load/store {full:.2}, lanes only {only:.2} (fired {fired} {fired2})");
    }
```
