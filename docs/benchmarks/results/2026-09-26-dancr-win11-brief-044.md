# usage: session.ps1 -Base <base binary> -Change <change binary> -Out <dir> -Root <builds' root> [-Parts criterion,census,one]
# The idle check (six consecutive five-second samples of the processor total below 15 per
# cent, none of the round's processes running), then the sampler, then the parts in order.
param([string]$Base, [string]$Change, [string]$Out, [string]$Root,
      [string[]]$Parts = @('criterion', 'census', 'one'))
$here = Split-Path -Parent $MyInvocation.MyCommand.Path
New-Item -ItemType Directory -Force $Out | Out-Null
$idle = Join-Path $Out 'idle.tsv'
"time`ttotal_pct" | Set-Content $idle
$ok = 0
for ($i = 0; $i -lt 4320; $i++) {
    $c = Get-Counter '\Processor(_Total)\% Processor Time' -SampleInterval 5 -MaxSamples 1
    $v = $c.CounterSamples[0].CookedValue
    "$($c.Timestamp.ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ss'))Z`t$([math]::Round($v, 1))" | Add-Content $idle
    if ($v -lt 15) { $ok++ } else { $ok = 0 }
    if ($ok -ge 6) { break }
}
if ($ok -lt 6) { "the machine was never idle for thirty seconds" | Add-Content (Join-Path $Out 'events.txt'); exit 3 }
"idle $((Get-Date).ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ss'))Z" | Add-Content (Join-Path $Out 'events.txt')
$sampler = Start-Process pwsh -ArgumentList @('-NoProfile', '-File', (Join-Path $here 'sampler.ps1'), '-Out', $Out, '-Root', $Root) -WindowStyle Hidden -PassThru
Start-Sleep -Seconds 12
$bash = 'C:\Program Files\Git\bin\bash.exe'
foreach ($part in $Parts) {
    & $bash (Join-Path $here 'pairs.sh').Replace('\', '/') $Base.Replace('\', '/') $Change.Replace('\', '/') $Out.Replace('\', '/') $part
}
Start-Sleep -Seconds 6
New-Item -ItemType File (Join-Path $Out 'stop') | Out-Null
$sampler.WaitForExit(30000) | Out-Null
"done $((Get-Date).ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ss'))Z" | Add-Content (Join-Path $Out 'events.txt')

`sampler.ps1` (the session ran it before `-Name` was added; its default is the pattern the session used):

```powershell
# usage: sampler.ps1 -Out <dir> -Root <directory the round's binaries are built under> [-Name <their process name pattern, active-* by default>]
# Every five seconds: the processor total over the five seconds, and from each process's
# processor time the logical processors held by the round's binaries (under -Root) and by every
# other process, with the three heaviest others. Stops when <Out>\stop exists.
param([string]$Out, [string]$Root, [string]$Name = 'active-*')
$log = Join-Path $Out 'load.tsv'
"time`ttotal_pct`tround_lp`tothers_lp`ttop_others" | Set-Content $log
$me = $PID
$prev = @{}
$sw = [Diagnostics.Stopwatch]::StartNew()
$last = $sw.Elapsed.TotalSeconds
for ($i = 0; $i -lt 100000; $i++) {
    if (Test-Path (Join-Path $Out 'stop')) { break }
    $c = Get-Counter '\Processor(_Total)\% Processor Time' -SampleInterval 5 -MaxSamples 1
    $total = $c.CounterSamples[0].CookedValue
    $now = $sw.Elapsed.TotalSeconds
    $dt = $now - $last
    $last = $now
    $cur = @{}
    $round = 0.0
    $others = @{}
    foreach ($p in Get-Process) {
        $t = $null
        try { $t = $p.TotalProcessorTime.TotalSeconds } catch { }
        if ($null -eq $t) { continue }
        $key = "$($p.Id):$($p.Name)"
        $cur[$key] = $t
        $d = if ($prev.ContainsKey($key)) { $t - $prev[$key] } else { $t }
        if ($i -eq 0) { continue }
        $isRound = ($p.Id -eq $me) -or ($p.Name -like $Name -and $p.Path -and $p.Path.StartsWith($Root))
        if ($isRound) { $round += $d } else { $others[$key] = $d }
    }
    $prev = $cur
    if ($i -eq 0) { continue }
    $sum = ($others.Values | Measure-Object -Sum).Sum
    $top = ($others.GetEnumerator() | Sort-Object Value -Descending | Select-Object -First 3 |
        ForEach-Object { "$($_.Key)=$([math]::Round($_.Value / $dt, 2))" }) -join ' '
    $stamp = $c.Timestamp.ToUniversalTime().ToString('yyyy-MM-ddTHH:mm:ss') + 'Z'
    "$stamp`t$([math]::Round($total, 1))`t$([math]::Round($round / $dt, 2))`t$([math]::Round($sum / $dt, 2))`t$top" |
        Add-Content $log
}
```

The rule, as `analyse.mjs` applies it:

```js
// A sample covers the five seconds before its stamp; it is in a part if it overlaps it.
const LIMIT = 4; // logical processors held by processes outside the round's binaries
const RUN = 3; // consecutive samples
function judge(part) {
  const w = window[part];
  if (!w || !w.end) return { part, complete: false };
  const s = load.filter((x) => x.t > w.start && x.t - 5000 < w.end);
  let run = 0, worst = 0, at = null;
  for (const x of s) {
    run = x.others > LIMIT ? run + 1 : 0;
    if (run > worst) { worst = run; at = x; }
  }
  const mean = (a) => a.reduce((p, q) => p + q, 0) / (a.length || 1);
  const totals = s.map((x) => x.total).sort((a, b) => a - b);
  return {
    part, complete: true, start: new Date(w.start).toISOString(), end: new Date(w.end).toISOString(),
    samples: s.length, disturbed: worst >= RUN, longestRunAboveLimit: worst, where: at ? `${new Date(at.t).toISOString()} ${at.top}` : null,
    total: { mean: +mean(totals).toFixed(1), max: totals.at(-1), atOrAbove40: totals.filter((v) => v >= 40).length },
    others: { mean: +mean(s.map((x) => x.others)).toFixed(2), max: Math.max(...s.map((x) => x.others)) },
    round: { mean: +mean(s.map((x) => x.round)).toFixed(2) },
  };
}
```
